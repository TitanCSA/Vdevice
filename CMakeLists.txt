cmake_minimum_required(VERSION 3.24)
project(Vdevice VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use Windows x64 and Visual Studio 2022 C++ tools.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc)
add_compile_definitions(UNICODE _UNICODE)
add_link_options(/PDBALTPATH:%_PDB%)
add_library(VdeviceSDK STATIC Vdevice/Vdevice.cpp)
add_library(Vdevice::SDK ALIAS VdeviceSDK)
target_include_directories(VdeviceSDK PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/Vdevice")
set_target_properties(VdeviceSDK PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin/SDK")
add_executable(VdeviceSetupNative InstallerConsole/Native/Setup.cpp)
target_link_libraries(VdeviceSetupNative PRIVATE setupapi newdev cfgmgr32 advapi32 wintrust crypt32)
target_link_options(VdeviceSetupNative PRIVATE /MANIFESTUAC:level='requireAdministrator')
set_target_properties(VdeviceSetupNative PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin/Native")
add_library(VdeviceBridge SHARED ExamplesUI/NativeBridge.cpp)
target_link_libraries(VdeviceBridge PRIVATE Vdevice::SDK)
set_target_properties(VdeviceBridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin/ExamplesUI")
file(GLOB sources CONFIGURE_DEPENDS "ExamplesConsole/*.cpp")
foreach(source IN LISTS sources)
  get_filename_component(name "${source}" NAME_WE)
  add_executable("${name}" "${source}")
  target_link_libraries("${name}" PRIVATE Vdevice::SDK advapi32)
  set_target_properties("${name}" PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin/ExamplesConsole")
endforeach()
foreach(name Keyboard Mouse Xbox360 DualShock4 XboxOneGip)
  add_executable("Vdevice${name}UI" WIN32 "ExamplesConsole/${name}.cpp")
  target_compile_definitions("Vdevice${name}UI" PRIVATE VDEVICE_EXAMPLE_UI)
  target_link_libraries("Vdevice${name}UI" PRIVATE Vdevice::SDK advapi32 user32 gdi32 comctl32 shell32)
  set_target_properties("Vdevice${name}UI" PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin/ExamplesUI")
endforeach()
enable_testing()
add_executable(ExampleTests Tests/ExampleTests.cpp)
target_link_libraries(ExampleTests PRIVATE Vdevice::SDK)
set_target_properties(ExampleTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin/Tests")
add_test(NAME ExampleProtocols COMMAND ExampleTests)
add_executable(InstallerNativeTests Tests/InstallerNativeTests.cpp)
target_link_libraries(InstallerNativeTests PRIVATE setupapi newdev cfgmgr32 advapi32 wintrust crypt32)
set_target_properties(InstallerNativeTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin/Tests")
add_test(NAME InstallerNative COMMAND InstallerNativeTests "${CMAKE_CURRENT_SOURCE_DIR}/InstallerUI/Package/Vdevice.inf")
foreach(name Keyboard Mouse Xbox360 DualShock4 XboxOneGip)
  add_test(NAME "${name}Descriptors" COMMAND "${name}" --validate)
  add_test(NAME "${name}Help" COMMAND "${name}" --help)
endforeach()
add_executable(ExampleLiveTests Tests/ExampleLiveTests.cpp)
target_link_libraries(ExampleLiveTests PRIVATE Vdevice::SDK advapi32 user32 setupapi hid xinput)
set_target_properties(ExampleLiveTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin/Tests")
# The live test is opt-in inside a Hyper-V guest, never an automatic CTest.
